def load_image_gray(path: Path):
    img = skio.imread(path)
    if img.ndim == 2:
        gray = img.astype(np.float32)
        if gray.max() > 1.01:
            gray = gray/255.0
    else:
        if img.shape[-1] == 4:
            img = img[..., :3]
        gray = rgb2gray(img).astype(np.float32)
    return np.clip(gray, 0.0, 1.0)

def angle(dx, dy):
    return np.mod(np.arctan2(dy, dx), np.pi)

def grad_orientation_hist(gray: np.ndarray, nbins: int = NBINS):
    dx = filters.sobel_h(gray)
    dy = filters.sobel_v(gray)
    theta = angle(dx, dy)
    counts, centers = exposure.histogram(theta, nbins=nbins)
    counts = counts.astype(np.float32)
    if counts.sum() > 0:
        counts = counts/(counts.sum()+1e-10)
    return counts

def list_images_by_class(data_dir: Path, classes):
    mapping = {}
    for cls in classes:
        cls_dir = data_dir / cls
        files = []
        if cls_dir.exists():
            for p in cls_dir.rglob('*'):
                if p.suffix.lower() in IMG_EXTS and p.is_file():
                    files.append(p)
        mapping[cls] = sorted(files)
    return mapping

def build_feature_matrix(images_by_class):
    X, y, paths = [], [], []
    for cls, files in images_by_class.items():
        for p in files:
            try:
                g = load_image_gray(p)
                h = grad_orientation_hist(g)
                X.append(h); y.append(cls); paths.append(str(p))
            except Exception as e:
                print(f'[WARN] skip {p}: {e}')
    X = np.vstack(X) if X else np.zeros((0, NBINS), dtype=np.float32)
    y = np.array(y)
    return X, y, paths

from sklearn.metrics import accuracy_score, f1_score, classification_report, confusion_matrix, ConfusionMatrixDisplay
from sklearn.model_selection import StratifiedKFold, cross_val_score

def evaluate_three(X_train, X_test, y_train, y_test, class_names, out_prefix='pa2'):
    classifiers = {
        'GaussianNB': GaussianNB(),
        'MLP_3x10': MLPClassifier(hidden_layer_sizes=(10,10,10), random_state=42, max_iter=500),
        'RandomForest': RandomForestClassifier(random_state=42)
    }
    results = []
    skf = StratifiedKFold(n_splits=5, shuffle=True, random_state=42)
    for name, clf in classifiers.items():
        print(f'\n--- {name} ---')
        cv_scores = cross_val_score(clf, X_train, y_train, cv=skf, scoring='accuracy', n_jobs=-1)
        print(f'Mean val acc (5-fold stratified): {cv_scores.mean():.4f} ± {cv_scores.std():.4f}')
        clf.fit(X_train, y_train)
        y_pred = clf.predict(X_test)
        acc = accuracy_score(y_test, y_pred)
        f1m = f1_score(y_test, y_pred, average='macro')
        print(f'Test accuracy: {acc:.4f} | Test macro-F1: {f1m:.4f}')
        print(classification_report(y_test, y_pred, digits=4))

        cm = confusion_matrix(y_test, y_pred, labels=class_names)
        disp = ConfusionMatrixDisplay(confusion_matrix=cm, display_labels=class_names)
        fig, ax = plt.subplots(figsize=(6,5))
        disp.plot(ax=ax)
        ax.set_title(f'Confusion Matrix — {name}')
        plt.tight_layout()
        cm_file = f'{out_prefix}_confusion_{name}.png'
        fig.savefig(cm_file); plt.show()
        print('Saved:', cm_file)

        dump(clf, f'{out_prefix}_{name}.joblib'); print('Saved model:', f'{out_prefix}_{name}.joblib')
        results.append({'method': name, 'mean_val_accuracy': float(cv_scores.mean()),
                        'test_accuracy': float(acc), 'test_macro_f1': float(f1m),
                        'confusion_matrix_file': cm_file, 'model_file': f'{out_prefix}_{name}.joblib'})
    df = pd.DataFrame(results)
    df.to_csv(f'{out_prefix}_summary.csv', index=False)
    print('Summary saved to', f'{out_prefix}_summary.csv')
    return df

from sklearn.model_selection import KFold, StratifiedKFold

def svm_two_class_model_selection(X_train_all, y_train_all, X_test_all, y_test_all,
                                  class_pairs=('Clean','Dusty'), Cs=[0.1,1,10,100], out_prefix='svm_select'):
    mask_tr = np.isin(y_train_all, class_pairs)
    mask_te = np.isin(y_test_all, class_pairs)
    Xtr, ytr = X_train_all[mask_tr], y_train_all[mask_tr]
    Xte, yte = X_test_all[mask_te], y_test_all[mask_te]
    if len(np.unique(ytr)) < 2:
        raise RuntimeError('Pick two classes that exist in training split.')

    kf  = KFold(n_splits=5, shuffle=True, random_state=42)
    skf = StratifiedKFold(n_splits=5, shuffle=True, random_state=42)

    train_k, val_k, train_sk, val_sk = [], [], [], []
    for C in Cs:
        tr_e, va_e = [], []
        for tr_idx, va_idx in kf.split(Xtr):
            clf = LinearSVC(C=C, max_iter=10000); clf.fit(Xtr[tr_idx], ytr[tr_idx])
            tr_e.append(1.0 - clf.score(Xtr[tr_idx], ytr[tr_idx]))
            va_e.append(1.0 - clf.score(Xtr[va_idx], ytr[va_idx]))
        train_k.append(np.mean(tr_e)); val_k.append(np.mean(va_e))

        tr_e, va_e = [], []
        for tr_idx, va_idx in skf.split(Xtr, ytr):
            clf = LinearSVC(C=C, max_iter=10000); clf.fit(Xtr[tr_idx], ytr[tr_idx])
            tr_e.append(1.0 - clf.score(Xtr[tr_idx], ytr[tr_idx]))
            va_e.append(1.0 - clf.score(Xtr[va_idx], ytr[va_idx]))
        train_sk.append(np.mean(tr_e)); val_sk.append(np.mean(va_e))

    import matplotlib.pyplot as plt
    plt.figure(figsize=(8,6))
    plt.plot(Cs, np.array(train_k)*100, marker='o', label='train error (KFold)')
    plt.plot(Cs, np.array(val_k)*100, marker='o', label='val error (KFold)')
    plt.plot(Cs, np.array(train_sk)*100, marker='s', label='train error (StratifiedKFold)')
    plt.plot(Cs, np.array(val_sk)*100, marker='s', label='val error (StratifiedKFold)')
    plt.xscale('log'); plt.xlabel('C (log scale)'); plt.ylabel('Mean error (%)')
    plt.title(f'SVM model selection: {class_pairs[0]} vs {class_pairs[1]}') if False else None
    # Explicit title since f-string above was conditional demo
    plt.title('SVM model selection: Clean vs Dusty')
    plt.legend(); plt.grid(True); plt.tight_layout()
    plot_file = f'{out_prefix}_error_curves.png'
    plt.savefig(plot_file); plt.show()
    print('Saved:', plot_file)

    best_idx = int(np.argmin(val_sk)); best_C = Cs[best_idx]
    print(f'Best C (Stratified 5-fold): {best_C} | mean val error = {val_sk[best_idx]:.4f}')
    clf_best = LinearSVC(C=best_C, max_iter=10000).fit(Xtr, ytr)
    test_err = 1.0 - clf_best.score(Xte, yte)
    print(f'Two-class test error (C={best_C}): {test_err:.4f}')
    return dict(Cs=Cs, train_errors_k=train_k, val_errors_k=val_k,
                train_errors_sk=train_sk, val_errors_sk=val_sk,
                best_C=best_C, test_error=test_err, plot_file=plot_file)

Step 8 — Analysis & Discussion¶

I tried three different models — Naïve Bayes, Neural Network (MLP 3×10), and Random Forest — and the Random Forest Classifier performed best. It had the highest validation and test accuracy (~66%), proving to be more stable and reliable for this dataset.

The Neural Network performed slightly worse, but still showed improvement compared to Naïve Bayes. Its simple design probably meant it didn't perform as well as it could have. With more layers or better tuning, it could have done better.

Naïve Bayes was the least accurate, as it confused similar categories like Clean and Dusty, because it made assumptions that didn't match the relationships seen in the data.

Overall, Random Forest is the most effective and balanced model here. It works well in most situations and doesn't need any complicated adjustments. To improve the MLP in future, its structure could be made deeper or it could be given more data.
